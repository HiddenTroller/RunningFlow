# 러닝플로우 별도 수정본 코드와 테스트

## 1 수정 계획 요약

현재 구조: main.py는 로컬 HTTP 요청을 분기하고, Studio는 사진을 분석합니다. 식약처 클라이언트는 제품과 주성분을 조회합니다. 기존 화면은 제품명만 연결하고 사람이 함량을 확인했습니다. 계산기는 단위를 환산하고 성분을 합산합니다. 필요한 파일은 모두 있어 추가 요청 없이 수정본을 만들었습니다.

| 파일 | 변경과 이유 |
|---|---|
| studio_client.py | 전체 노드 요청, Extract 우선, config_id 생략으로 항상 최신 설정 요청, finally 파일 삭제 |
| matcher.py | 이름·함량 대조와 5가지 상태, 공식 값만 계산 제품으로 변환 |
| normalization.py | 이름 정규화 공유 및 모듈 순환 참조 방지 |
| workflow.py | 후보 정렬, 확인 기록, 공식 계산 스냅샷 관리 |
| mfds_client.py | 검색 페이지 크기와 제조사 필터 옵션 추가 |
| main.py | 자동 대조·직접 재검색·확인 경로, 테스트 가능한 요청 분기 |
| static/ | 후보 대조 표와 불일치 추가 확인, 공식값 기반 계산 요청 |
| calculator.py | 공통 mg 환산, 코드 없는 이름 합산, product_codes 선택 조건 |
| config.py | 요청에 맞춰 인증키를 .env에서만 읽음 |

원본 프로젝트는 유지됩니다. 새 프로젝트의 포트는 8767입니다. 실제 키는 복사하거나 문서에 포함하지 않았습니다.

**테스트 충돌 처리:** 원본 27개는 그대로 통과합니다. 수정본의 기존 27개 중 공백 차이를 합치지 않는 검사와 설정 ID를 무시하는 검사 두 개는 새 요구사항과 반대이므로 데이터/기대값을 변경했습니다. 자세한 기록은 README의 기존 테스트 변경 기록과 원본대비변경.patch를 보세요.

```python
"""Studio Extract 우선 선택, 항상 최신 설정 정책, 업로드 파일 정리."""
import json
import logging
import re
import time
import uuid
from urllib.parse import quote
from urllib.request import Request, urlopen
from urllib.error import HTTPError, URLError

from config import settings
from mfds_client import ServiceError
from extraction_adapter import normalize_extraction

BASE = 'https://api.upstage.ai/v2'
LOG = logging.getLogger(__name__)


def build_job_payload(agent, file_id, config_id=None):
    """항상 최신 설정. config_id 인자는 과거 호출 호환용이며 의도적으로 무시합니다."""
    payload = {'model': agent, 'include': ['all'], 'input': [
        {'role': 'user', 'content': [{'type': 'input_file', 'file_id': file_id}]}]}
    return payload


def _request(path, key, payload=None, content_type='application/json', method=None):
    request = Request(BASE + path, data=payload,
                      headers={'Authorization': 'Bearer ' + key, 'Content-Type': content_type}, method=method)
    try:
        with urlopen(request, timeout=45) as response:
            raw = response.read(16 * 1024 * 1024 + 1)
            if len(raw) > 16 * 1024 * 1024:
                raise ServiceError('Studio 결과 크기를 초과했습니다. 확인 필요.')
            return json.loads(raw) if raw else {}
    except HTTPError as error:
        raise ServiceError(f'Upstage HTTP {error.code}: 키·접근 권한·사용 한도를 확인하세요.') from None
    except (URLError, TimeoutError):
        raise ServiceError('Upstage에 연결하지 못했습니다. 확인 필요.') from None
    except (ValueError, TypeError, UnicodeError):
        raise ServiceError('Upstage 응답 형식을 확인할 수 없습니다.') from None


def _step_text(step):
    if not isinstance(step, dict):
        return ''
    parts = step.get('content')
    if not isinstance(parts, list):
        return ''
    return '\n'.join(p['text'] for p in parts if isinstance(p, dict)
                     and p.get('type') == 'output_text' and isinstance(p.get('text'), str))


def _parse_json(text):
    text = text.strip()
    fenced = re.fullmatch(r'```(?:json)?\s*([\s\S]*?)\s*```', text, re.I)
    if fenced:
        text = fenced.group(1)
    try:
        parsed = json.loads(text)
        return parsed if isinstance(parsed, dict) else None
    except (ValueError, TypeError):
        return None


def select_output(job):
    outputs = job.get('output')
    outputs = outputs if isinstance(outputs, list) else []
    extracts = [s for s in outputs if isinstance(s, dict) and 'extract' in str(s.get('model', '')).lower()]
    for step in reversed(extracts):
        text = _step_text(step)
        parsed = _parse_json(text)
        if parsed is not None:
            return text, parsed, str(step.get('model')), []
    warnings = ['Extract 결과를 JSON으로 읽지 못해 마지막 출력으로 대체했습니다. 확인 필요.'] if extracts else []
    step = outputs[-1] if outputs else {}
    text = _step_text(step) or (job.get('output_text') if isinstance(job.get('output_text'), str) else '')
    return text, _parse_json(text), str(step.get('model', 'last')) if isinstance(step, dict) else 'last', warnings


def analyze_photo(content, mime):
    config = settings()
    key, agent = config.get('UPSTAGE_API_KEY', ''), config.get('UPSTAGE_AGENT_ID', '')
    if not key or not agent.startswith('agt_'):
        raise ServiceError('.env에 UPSTAGE_API_KEY와 UPSTAGE_AGENT_ID를 입력하세요.')
    extension = {'image/jpeg': 'jpg', 'image/png': 'png', 'image/webp': 'webp'}.get(mime)
    if not extension or not content or len(content) > 10 * 1024 * 1024:
        raise ServiceError('JPG·PNG·WEBP 사진 한 장(10MB 이하)을 선택하세요.')
    boundary = 'Learningflow' + uuid.uuid4().hex
    body = (f'--{boundary}\r\nContent-Disposition: form-data; name="purpose"\r\n\r\nuser_data\r\n'
            f'--{boundary}\r\nContent-Disposition: form-data; name="file"; filename="photo.{extension}"\r\n'
            f'Content-Type: {mime}\r\n\r\n').encode() + content + f'\r\n--{boundary}--\r\n'.encode()
    file_id, result = None, None
    try:
        uploaded = _request('/files', key, body, 'multipart/form-data; boundary=' + boundary)
        file_id = uploaded.get('id') if isinstance(uploaded, dict) else None
        if not file_id or not isinstance(file_id, str):
            raise ServiceError('Upstage 파일 ID를 받지 못했습니다.')
        payload = build_job_payload(agent, file_id)
        job = _request('/responses', key, json.dumps(payload).encode())
        if not isinstance(job, dict) or not job.get('id'):
            raise ServiceError('Upstage 작업 ID를 받지 못했습니다.')
        deadline = time.monotonic() + 240
        while job.get('status') in ('queued', 'in_progress'):
            if time.monotonic() >= deadline:
                raise ServiceError('분석 대기 시간이 초과되었습니다. Studio 작업 상태를 확인하세요.')
            time.sleep(2)
            job = _request('/responses/' + quote(str(job['id']), safe='') + '?include=all', key)
        if job.get('status') != 'completed':
            raise ServiceError('Studio 작업이 완료되지 않았습니다. 실패 노드를 확인하세요.')
        text, raw_extracted, selected, warnings = select_output(job)
        extracted, adapter_warnings = normalize_extraction(raw_extracted)
        warnings.extend(adapter_warnings)
        result = {'job_id': job['id'], 'agent_id': agent,
                  'config_policy': 'latest',
                  'requested_config_id': None, 'config_id': job.get('config_id'),
                  'legacy_config_ignored': bool(config.get('UPSTAGE_CONFIG_ID', '').strip()),
                  'selected_model': selected, 'text': text, 'extracted': extracted,
                  'raw_extracted': raw_extracted, 'warnings': warnings}
        return result
    finally:
        if file_id:
            try:
                deleted = _request('/files/' + quote(file_id, safe=''), key, method='DELETE')
                if isinstance(deleted, dict) and deleted.get('deleted') is False:
                    raise ServiceError('파일 삭제 미완료')
                if result is not None:
                    result['file_cleanup'] = 'deleted'
            except Exception:
                message = 'Upstage 업로드 파일 삭제를 완료하지 못했습니다. Studio에서 파일 보관 상태를 확인하세요.'
                LOG.warning(message)
                if result is not None:
                    result['warnings'].append(message)
                    result['file_cleanup'] = 'warning'

```

### matcher.py

```python
"""사진 Extract와 공식 주성분을 대조. 추측·동의어 치환·부동소수점 허용오차 없음."""
import json
import re
from decimal import Decimal

from calculator import to_mg, number
from mfds_client import ServiceError
from normalization import normalize_name

UNITS = {'mg': 'mg', '밀리그램': 'mg', 'g': 'g', '그램': 'g',
         'μg': 'μg', 'µg': 'μg', 'ug': 'μg', 'mcg': 'μg', '마이크로그램': 'μg'}
UNIT_PATTERN = '|'.join(sorted(map(re.escape, UNITS), key=len, reverse=True))
BASIS = r'(?P<basis>[0-9]+(?:\.[0-9]+)?)\s*(?P<basis_unit>정|캡슐|포|mL|ml)'
STRENGTH = re.compile(r'^\s*(?:' + BASIS + r'\s*중\s*)?(?P<amount>[0-9]+(?:\.[0-9]+)?)\s*'
                      r'(?P<unit>' + UNIT_PATTERN + r')\s*$', re.I)


def text(value):
    return value.strip() if isinstance(value, str) else ''


def parse_strength(value):
    result = {'raw': text(value), 'mg': None, 'amount': None, 'unit': None,
              'basis_quantity': None, 'basis_unit': None, 'status': '함량 미확인'}
    if not isinstance(value, str) or len(value) > 300:
        return result
    match = STRENGTH.fullmatch(value)
    if not match:
        return result
    groups = match.groupdict()
    raw_unit = groups['unit'].lower()
    unit = UNITS.get(raw_unit)
    try:
        if not unit:
            return result
        mg = to_mg(groups['amount'], unit)
        basis = number(groups['basis'], '기준 수량') if groups['basis'] else None
    except ServiceError:
        return result
    result.update(mg=str(mg), amount=groups['amount'], unit=unit, status='확인됨',
                  basis_quantity=str(basis) if basis is not None else None,
                  basis_unit='mL' if str(groups['basis_unit']).lower() == 'ml' else groups['basis_unit'])
    return result


def parse_basis(value):
    """명시된 1정/100mL 등만 사용. 1정(총량 700mg) 중 표현도 처리."""
    value = text(value)
    pattern = r'([0-9]+(?:\.[0-9]+)?)\s*(정|캡슐|포|mL|ml)(?:\s*\([^()]*\))?\s*(?:중|당)'
    found = re.findall(pattern, value, re.I)
    if not found:
        exact = re.fullmatch(r'\s*([0-9]+(?:\.[0-9]+)?)\s*(정|캡슐|포|mL|ml)\s*', value, re.I)
        found = [exact.groups()] if exact else []
    normalized = {(str(Decimal(q)), 'mL' if u.lower() == 'ml' else u) for q, u in found}
    if len(normalized) != 1:
        return None, None
    quantity, unit = normalized.pop()
    try:
        return str(number(quantity, '공식 기준 수량')), unit
    except ServiceError:
        return None, None


def photo_ingredients(extracted):
    extracted = extracted if isinstance(extracted, dict) else {}
    values = extracted.get('effective_ingredients')
    if not isinstance(values, list):
        return []
    result = []
    for row in values[:100]:
        if not isinstance(row, dict):
            continue
        name = text(row.get('ingredient_name')) or text(row.get('base_ingredient_name'))
        aliases = {normalize_name(row.get('ingredient_name')), normalize_name(row.get('base_ingredient_name'))} - {''}
        result.append({'name': name, 'aliases': sorted(aliases),
                       'strength': parse_strength(row.get('strength_per_unit')), 'raw': row})
    return result


def official_ingredients(detail):
    product = detail.get('product') if isinstance(detail.get('product'), dict) else {}
    values = detail.get('ingredients_raw')
    result = []
    for row in values if isinstance(values, list) else []:
        if not isinstance(row, dict):
            continue
        name = text(row.get('MTRAL_NM'))
        qnt, raw_unit = row.get('QNT'), text(row.get('INGD_UNIT_CD'))
        qnt = str(qnt) if isinstance(qnt, (str, int, float)) and not isinstance(qnt, bool) else ''
        strength = parse_strength(qnt + raw_unit)
        # 코드형 단위나 불명확한 분량은 원문을 임의 해석하지 않음.
        bases = [parse_basis(row.get('CPNT_CTNT_CONT')), parse_basis(product.get('TOTAL_CONTENT'))]
        known = {b for b in bases if b[0] is not None}
        basis = next(iter(known)) if len(known) == 1 else (None, None)
        result.append({'name': name, 'normalized_name': normalize_name(name),
                       'ingredient_code': text(row.get('MTRAL_CODE')),
                       'strength': strength, 'basis_quantity': basis[0], 'basis_unit': basis[1], 'raw': row})
    return result


def compare_ingredients(extracted, detail):
    photos, official = photo_ingredients(extracted), official_ingredients(detail)
    remaining = set(range(len(official)))
    rows, warnings = [], list(detail.get('warnings') or [])
    for photo in photos:
        matches = [i for i in remaining if official[i]['normalized_name']
                   and official[i]['normalized_name'] in photo['aliases']]
        if len(matches) != 1:
            rows.append({'name': photo['name'] or '성분명 미확인', 'status': '사진에만 있음',
                         'photo': photo, 'official': None})
            if len(matches) > 1:
                warnings.append('동일 이름의 공식 성분 행이 여러 개입니다. 확인 필요.')
            continue
        i = matches[0]; remaining.remove(i); target = official[i]
        a, b = photo['strength'], target['strength']
        status = '함량 미확인'
        if a['mg'] is not None and b['mg'] is not None:
            # 양쪽에 기준이 명시되면 기준 수량당 mg로 비교. 기준 단위가 다르면 비교 불가.
            if a['basis_quantity'] and target['basis_quantity']:
                if a['basis_unit'] == target['basis_unit']:
                    amg = Decimal(a['mg']) / Decimal(a['basis_quantity'])
                    bmg = Decimal(b['mg']) / Decimal(target['basis_quantity'])
                    status = '일치' if amg == bmg else '함량 불일치'
            elif ((a['basis_quantity'] and Decimal(a['basis_quantity']) != 1)
                  or (target['basis_quantity'] and Decimal(target['basis_quantity']) != 1)):
                # 어느 쪽이든 비단위 기준인데 상대 기준이 없으면 같은 분모라고 가정하지 않음.
                status = '함량 미확인'
            else:
                status = '일치' if Decimal(a['mg']) == Decimal(b['mg']) else '함량 불일치'
        rows.append({'name': target['name'], 'status': status, 'photo': photo, 'official': target})
    for i in sorted(remaining):
        rows.append({'name': official[i]['name'] or '성분명 미확인', 'status': '공식에만 있음',
                     'photo': None, 'official': official[i]})
    raw_photo_rows = extracted.get('effective_ingredients') if isinstance(extracted, dict) else None
    malformed_photo = (not isinstance(raw_photo_rows, list) or len(raw_photo_rows) > 100
                       or any(not isinstance(r, dict) for r in raw_photo_rows))
    if malformed_photo:
        warnings.append('사진의 성분 목록 형식이 불완전합니다. 확인 필요.')
    all_match = bool(rows) and all(r['status'] == '일치' for r in rows) and not malformed_photo
    calculation_ready = bool(official) and all(o['name'] and o['strength']['mg'] is not None
                                             and o['basis_quantity'] and o['basis_unit'] for o in official)
    if not calculation_ready:
        warnings.append('공식 성분·함량·함량 기준을 완전히 확인하지 못했습니다. 계산 추가 차단.')
    if len({(o['ingredient_code'] or o['normalized_name']) for o in official}) != len(official):
        calculation_ready = False
        warnings.append('공식 성분이 중복된 행으로 반환되었습니다. 계산 추가 차단.')
    if detail.get('warnings'):
        calculation_ready = False
    return {'status': '일치' if all_match else '확인 필요', 'all_match': all_match,
            'rows': rows, 'official_ingredients': official, 'calculation_ready': calculation_ready,
            'warnings': warnings}


def manufacturer_matches(photo_name, official_name):
    a, b = normalize_name(photo_name), normalize_name(official_name)
    return bool(a and b and a == b)


def official_calculation_product(detail, comparison, extracted):
    """계산값은 공식 데이터에서만 생성. 사진은 evidence 안에만 남김."""
    if not comparison.get('calculation_ready'):
        raise ServiceError('공식 함량 또는 기준 단위가 미확인입니다. 확인 필요.')
    product = detail['product']
    ingredients = []
    for o in comparison['official_ingredients']:
        ingredients.append({'name': o['name'], 'ingredient_code': o['ingredient_code'],
                            'amount': o['strength']['amount'], 'unit': o['strength']['unit'],
                            'basis_quantity': o['basis_quantity'], 'basis_unit': o['basis_unit'],
                            'evidence': json.dumps({'official': o['raw'], 'photo': extracted}, ensure_ascii=False)})
    units = {i['basis_unit'] for i in ingredients}
    if len(units) != 1:
        raise ServiceError('공식 성분의 기준 단위가 다릅니다. 확인 필요.')
    return {'name': text(product.get('ITEM_NAME')), 'code': str(product.get('ITEM_SEQ', '')),
            'quantity': '0', 'quantity_unit': next(iter(units)), 'confirmed': True,
            'source': detail.get('source', ''), 'ingredients': ingredients,
            'warning_text': text(product.get('NB_DOC_DATA')), 'directions_text': text(product.get('UD_DOC_DATA'))}
```

### normalization.py

```python
"""표기만 정리합니다. 염형·동의어·다른 화학명은 변환하지 않습니다."""
import re
import unicodedata

STANDARD = re.compile(r'^(?:kp|usp(?:-?nf)?|ep|bp|jp|ph\.?eur\.?|별규|약전|규격)[\s.\-\d]*$', re.I)


def normalize_name(value):
    if not isinstance(value, str):
        return ''
    value = unicodedata.normalize('NFKC', value).lower()
    value = re.sub(r'\(([^()]*)\)', lambda m: '' if STANDARD.fullmatch(m.group(1).strip()) else m.group(0), value)
    return re.sub(r'\s+', '', value)

```

### workflow.py

```python
"""대조·후보 정렬·명시적 확인과 공식 데이터 계산 경계를 담당합니다."""
import copy
import secrets
import threading
import time

from matcher import compare_ingredients, manufacturer_matches, official_calculation_product, text
from mfds_client import MFDSClient, ServiceError
from studio_client import analyze_photo
from calculator import calculate
from extraction_adapter import normalize_extraction


class AnalysisWorkflow:
    def __init__(self, client=None, analyzer=None):
        self.client = client or MFDSClient()
        self.analyzer = analyzer or analyze_photo
        self.records = {}
        self.lock = threading.Lock()

    def _put(self, kind, value):
        with self.lock:
            now = time.monotonic()
            self.records = {k: v for k, v in self.records.items() if now - v[0] < 1800}
            if len(self.records) >= 300:
                oldest = min(self.records, key=lambda k: self.records[k][0])
                del self.records[oldest]
            token = secrets.token_urlsafe(24)
            self.records[token] = (now, kind, copy.deepcopy(value))
            return token

    def _get(self, token, kind):
        if not isinstance(token, str):
            raise ServiceError('확인 정보가 없습니다. 후보를 다시 조회하세요.')
        with self.lock:
            record = self.records.get(token)
            if not record or record[1] != kind or time.monotonic() - record[0] >= 1800:
                raise ServiceError('확인 정보가 만료되었습니다. 후보를 다시 조회하세요.')
            return copy.deepcopy(record[2])

    def analyze_and_match(self, content, mime):
        studio = self.analyzer(content, mime)

    def match_extracted(self, extracted, name=None, page=1):
        extracted, extraction_warnings = normalize_extraction(extracted)
        photo_id = self._put('photo', extracted)

        if not name:
            return {'status': '제품을 찾지 못함', 'reason': '사진의 제품명이 미확인입니다. 직접 검색하세요.',

        # 후보별 상세 호출 폭증을 막고, 일부만 평가되면 명확히 알립니다.
        limit = 10
        seen = set()
        for item in found['items'][:limit]:
            code = str(item.get('ITEM_SEQ', ''))
            if not code or code in seen:
                continue
            seen.add(code)
            try:
                detail = self.client.detail(code)
                comparison = compare_ingredients(extracted, detail)
            except ServiceError:
                detail = {'product': item, 'ingredients_raw': [], 'warnings': ['공식 상세 조회 실패. 확인 필요.'],
                          'source': found.get('source', '')}
                comparison = compare_ingredients(extracted, detail)

                comparison['status'] = '확인 필요'
                comparison['all_match'] = False
            manufacturer_match = manufacturer_matches(extracted.get('manufacturer'), detail['product'].get('ENTP_NAME'))
            record = {'detail': detail, 'comparison': comparison, 'extracted': extracted}
            candidate_id = self._put('candidate', record)
            candidates.append({'candidate_id': candidate_id, 'product': detail['product'],
                               'manufacturer_match': manufacturer_match,
                               'comparison': comparison, 'source': detail.get('source', '')})
        candidates.sort(key=lambda c: (not c['manufacturer_match'], not c['comparison']['all_match'],
                                       str(c['product'].get('ITEM_SEQ', ''))))
        if found['total'] > len(candidates):
            warnings.append('이번 페이지의 최대 10개 후보를 대조했습니다. 다른 후보는 제품명을 좁혀 검색하거나 다음 페이지에서 확인하세요.')
        return {'status': '후보 확인 필요' if candidates else '제품을 찾지 못함',
```

```python
    def do_POST(self):
        if (not self.valid_host() or self.headers.get('Origin') != 'http://' + ALLOWED_HOST
                or not hmac.compare_digest(self.headers.get('X-App-Token', ''), TOKEN)):
            self.send(403, {'error': '이 앱 화면에서 요청해 주세요.'})
            return
        try:
            length = int(self.headers.get('Content-Length', '0'))
            if not 0 < length <= 15 * 1024 * 1024:
                raise ServiceError('요청 크기가 허용 범위를 벗어났습니다.')
            data = json.loads(self.rfile.read(length))
            if not isinstance(data, dict):
                raise ServiceError('요청 형식을 확인하세요.')
            path = urlsplit(self.path).path
            result = dispatch_post(path, data)
            self.send(200, result)
        except ServiceError as error:
            self.send(400, {'error': str(error)})
        except (json.JSONDecodeError, ValueError, TypeError, KeyError):
            self.send(400, {'error': '입력 데이터 또는 로컬 기준 파일 형식을 확인하세요.'})
        except Exception:
            self.send(500, {'error': '처리 중 오류가 발생했습니다. 인증키를 제외한 입력 형식을 확인해 주세요.'})


def run():
    global ALLOWED_HOST
    parser = argparse.ArgumentParser(description='러닝플로우 약품 성분 확인')
    parser.add_argument('--no-browser', action='store_true')
    args = parser.parse_args()
    port = int(settings().get('PORT', '8767'))
    ALLOWED_HOST = f'127.0.0.1:{port}'
    try:
        server = ThreadingHTTPServer(('127.0.0.1', port), Handler)
    except OSError:
        raise SystemExit('앱을 열 수 없습니다. 다른 실행 창을 닫거나 .env의 PORT를 변경하세요.') from None
    print(f'러닝플로우 실행: http://{ALLOWED_HOST}')
    print('키는 .env에 입력하세요. 종료: Ctrl+C. 현재 PC에서만 접속할 수 있습니다.')
    if not args.no_browser:
        webbrowser.open('http://' + ALLOWED_HOST)
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        pass
    finally:
        server.server_close()


if __name__ == '__main__':
    run()
```

```python
"""확인된 입력만 계산합니다. 성분명 유사도 매칭 및 최대량 추정은 하지 않습니다."""
from decimal import Decimal, InvalidOperation
from datetime import datetime, timezone

from mfds_client import ServiceError
from normalization import normalize_name

FACTORS = {'mg': Decimal('1'), 'g': Decimal('1000'), 'μg': Decimal('0.001'),
           'µg': Decimal('0.001'), 'ug': Decimal('0.001'), 'mcg': Decimal('0.001')}


def number(value, label, allow_zero=False):
    try:
        result = Decimal(str(value))
    except (InvalidOperation, ValueError, TypeError):
        raise ServiceError(label + ': 숫자를 입력하세요.') from None
    if not result.is_finite() or result < 0 or (not allow_zero and result == 0) or result > 1000000:
        raise ServiceError(label + ': 유효한 범위의 양수를 입력하세요.')
    return result


def to_mg(value, unit):
    """대조와 계산에서 공통 Decimal 환산 로직 사용."""
    if unit not in FACTORS:
        raise ServiceError('지원하지 않는 함량 단위입니다. 확인 필요.')
    return number(value, '함량') * FACTORS[unit]


def calculate(products, mode='single', rules=None):
    if mode not in ('single', 'day'):
        raise ServiceError('1회 가정 또는 1일 가정을 선택하세요.')
    if not isinstance(products, list) or not products or len(products) > 30:
        raise ServiceError('계산할 제품을 1~30개 추가하세요.')
    totals, seen, exclusions = {}, set(), []
    for product in products:
        if not isinstance(product, dict):
            raise ServiceError('제품 데이터 형식이 잘못되었습니다.')
        name = str(product.get('name', '')).strip()
        code = str(product.get('code', '')).strip()
        if not name or not code or code in seen:
            raise ServiceError('제품명·품목코드를 확인하고 같은 제품을 여러 번 추가하지 마세요.')
        seen.add(code)
        if product.get('confirmed') is not True:
            raise ServiceError(name + ': 공식 원문과 함량 기준 확인이 필요합니다.')
        quantity = number(product.get('quantity'), name + ' 가정 수량', allow_zero=True)
        if quantity == 0:
            exclusions.append(name + ': 가정 수량 0으로 제외')
            continue
        basis_unit = product.get('quantity_unit')
        ingredients = product.get('ingredients')
        if basis_unit not in ('정', '캡슐', 'mL', '포') or not isinstance(ingredients, list) or not ingredients:
            raise ServiceError(name + ': 성분과 수량 단위를 확인하세요.')
        ingredient_seen = set()
        for ingredient in ingredients:
            if not isinstance(ingredient, dict):
                raise ServiceError(name + ': 성분 형식이 잘못되었습니다.')
            component = str(ingredient.get('name', '')).strip()
            mapping_code = str(ingredient.get('ingredient_code', '')).strip()
            unit = ingredient.get('unit')
            evidence = str(ingredient.get('evidence', '')).strip()
            if not component or not evidence or unit not in FACTORS or ingredient.get('basis_unit') != basis_unit:
                raise ServiceError(name + ': 성분명·지원 단위·함량 기준·원문 근거를 확인하세요.')
            # 코드가 있으면 코드로, 없으면 정확한 표기명으로만 묶고 일치 수준을 별도 표시합니다.
            key = 'code:' + mapping_code if mapping_code else 'name:' + normalize_name(component)
            if key in ingredient_seen:
                raise ServiceError(name + ': 동일 성분 행이 반복됩니다. 중복 행을 확인하세요.')
            ingredient_seen.add(key)
            amount = number(ingredient.get('amount'), component + ' 함량')
            basis = number(ingredient.get('basis_quantity'), component + ' 기준 수량')
            dose = to_mg(amount, unit) * quantity / basis
            row = totals.setdefault(key, {'ingredient': component, 'ingredient_code': mapping_code,
                                         'matching': '공식 성분코드' if mapping_code else '이름 기준으로 묶음',
                                         'total': Decimal('0'), 'products': [], 'contributions': []})
            row['total'] += dose
            row['products'].append(code)
            row['contributions'].append({'product': name, 'product_code': code, 'mg': str(dose),
                                         'evidence': evidence, 'source': product.get('source', ''),
                                         'quantity': str(quantity), 'quantity_unit': basis_unit})
    if not totals:
        raise ServiceError('0보다 큰 가정 수량을 하나 이상 입력하세요.')
    output = []
    for row in totals.values():
        row['total_mg'] = str(row.pop('total'))
        row['duplicate'] = len(row['products']) > 1
        # 기준은 서버 로컬의 검토 완료 규칙에서만 가져오며 클라이언트 최대량은 받지 않습니다.
        candidates = [r for r in (rules or []) if r.get('reviewed') is True
                      and r.get('ingredient_code') and r.get('ingredient_code') == row['ingredient_code']
                      and r.get('mode') == mode and r.get('population') == 'adult'
                      and r.get('reviewer') and r.get('reviewed_at') and r.get('source_url')
                      and (not r.get('product_codes') or
                           set(row['products']).issubset(set(r['product_codes'])))]
        row['criterion_status'] = 'unknown'
        row['criterion_note'] = '적용 가능한 검증 기준 없음. 초과 여부 미판정.'
        if len(candidates) == 1:
            rule = candidates[0]
            limit = number(rule.get('limit_mg'), '검증 기준')
            row.update(criterion_status='exceeded' if Decimal(row['total_mg']) > limit else 'not_exceeded',
                       limit_mg=str(limit), rule_source=rule['source_url'], rule_id=rule.get('id', ''),
                       criterion_note='등록된 성인 가정 시나리오의 용량 기준만 비교. 복용 허가 아님.')
        elif len(candidates) > 1:
            row['criterion_note'] = '적용 기준이 여러 개입니다. 기준 충돌 확인 필요.'
        output.append(row)
    return {'mode': mode, 'created_at': datetime.now(timezone.utc).isoformat(), 'ingredients': output,
            'overall': 'not_a_safety_assessment', 'excluded': exclusions,
            'unassessed': ['실제 복용 이력·시간·간격', '병용금기 및 효능군 중복의 자동 조회',
                           '개인 질환·임신·간·신장 기능', '성분코드가 없는 성분의 동의어·염형 차이'],
            'notice': '가정 수량에 따른 계산입니다. 최대량 미초과는 추가 복용 가능 또는 병용 안전을 의미하지 않습니다.'}
```

```python
from pathlib import Path

ROOT = Path(__file__).resolve().parent


def settings():
    values = {}
    path = ROOT / '.env'
    if path.exists():
        for line in path.read_text(encoding='utf-8-sig').splitlines():
            line = line.strip()
            if not line or line.startswith('#') or '=' not in line:
                continue
            key, value = line.split('=', 1)
            values[key.strip()] = value.strip().strip('\"\'')
    return values
```

```python
"""Studio의 한국어/영문 Extract를 공통 구조로 연결. 원문 보존, 정보 추정 금지."""
import re


def scalar(value):
    if isinstance(value, str):
        return value.strip()
    if isinstance(value, list):
        values = [v.strip() for v in value if isinstance(v, str) and v.strip()]
        # 제품명처럼 단일 값은 한 값이 명확할 때만 사용합니다.
        return values[0] if len(values) == 1 else ''
    return ''


def original_text(value):
    if isinstance(value, str):
        return value.strip()
    if isinstance(value, list):
        return '\n'.join(v.strip() for v in value if isinstance(v, str) and v.strip())
    return ''


def pick(data, *keys):
    for key in keys:
        value = data.get(key)
        if value not in (None, '', []):
            return value
    return None


def ingredient_row(value):
    if isinstance(value, dict):
        result = dict(value)
        name = scalar(pick(value, 'ingredient_name', '성분명'))
        strength = scalar(pick(value, 'strength_per_unit', '함량 원문'))
        if not strength:
            amount, unit = pick(value, '함량', 'amount'), scalar(pick(value, '단위', 'unit'))
            if isinstance(amount, (str, int, float)) and not isinstance(amount, bool) and unit:
                strength = str(amount).strip() + ' ' + unit
                basis = scalar(pick(value, '함량 기준', '함량기준', 'basis'))
                if re.fullmatch(r'[0-9]+(?:\.[0-9]+)?\s*(?:정|캡슐|포|mL)', basis, re.I):
                    strength = basis + ' 중 ' + strength
        result.update(ingredient_name=name,
                      base_ingredient_name=scalar(pick(value, 'base_ingredient_name', '기본성분명')) or name,
                      strength_per_unit=strength,
                      ingredient_standard=scalar(pick(value, 'ingredient_standard', '규격')),
                      source_raw=value.get('source_raw', value))
        return result
    if not isinstance(value, str) or not value.strip():
        return None
    raw = value.strip()
    parts = [part.strip() for part in raw.split('/')]
    name, strength = '', ''
    if len(parts) in (3, 4) and parts[0]:
        # 실제 저장 결과 예: 아세트아미노펜 / 500 / mg / 1정.
        # 숫자와 단위를 결합할 뿐, 해석 불가 함량을 유효 숫자로 만들지 않습니다.
        name = parts[0]
        strength = parts[1] + ' ' + parts[2]
        if len(parts) == 4:
            basis = parts[3]
            if re.fullmatch(r'[0-9]+(?:\.[0-9]+)?\s*(?:정|캡슐|포|mL)', basis, re.I):
                strength = basis + ' 중 ' + strength
            elif basis not in ('', '기재 없음', '판독 불가', '확인 필요'):
                # 알 수 없는 분모를 버려 잘못 일치시키지 않습니다.
                strength = raw
    else:
        # 사람이 보기 좋은 결합 표기도 명시된 숫자/단위가 있을 때만 분리.
        match = re.fullmatch(r'(.+?)\s+([0-9]+(?:\.[0-9]+)?\s*(?:밀리그램|마이크로그램|그램|mg|mcg|ug|μg|µg|g))(?:\s*\(([0-9]+(?:\.[0-9]+)?\s*(?:정|캡슐|포|mL))당\))?', raw, re.I)
        if match:
            name, strength = match.group(1).strip(), match.group(2)
            if match.group(3):
                strength = match.group(3) + ' 중 ' + strength
    return {'ingredient_name': name or raw, 'base_ingredient_name': name or raw,
            'strength_per_unit': strength, 'ingredient_standard': '', 'source_raw': value}


def normalize_extraction(data):
    if not isinstance(data, dict):
        return {}, ['Extract JSON 객체를 확인하지 못했습니다. 확인 필요.']
    # 한 단계의 표준 래퍼만 해제합니다. 임의 재귀 검색으로 다른 제품을 섞지 않음.
    for wrapper in ('extracted', 'data'):
        if not any(k in data for k in ('제품명', 'product_name_raw', 'effective_ingredients', '유효성분 목록')) and isinstance(data.get(wrapper), dict):
            data = data[wrapper]
            break
    warnings = list(data.get('adapter_warnings', [])) if isinstance(data.get('adapter_warnings'), list) else []
    result = dict(data)
    result.update(product_name_raw=scalar(pick(data, 'product_name_raw', '제품명', 'product_name')),
                  manufacturer=scalar(pick(data, 'manufacturer', '제조사', '업체명')),
                  dosage_and_administration=original_text(pick(data, 'dosage_and_administration', '성인 용법·용량', '성인 용법용량')),
                  max_daily_dose=original_text(pick(data, 'max_daily_dose', '하루 최대량')))
    values = pick(data, 'effective_ingredients', '유효성분 목록')
    if isinstance(values, str):
        values = re.split(r'(?:\r?\n|<br\s*/?>)', values, flags=re.I)
    if not isinstance(values, list):
        values = []
        warnings.append('사진의 유효성분 목록이 없거나 지원하지 않는 형식입니다. 확인 필요.')
    converted = []
    for value in values[:100]:
        row = ingredient_row(value)
        if row is not None:
            converted.append(row)
        else:
            warnings.append('사진의 일부 성분 행을 읽지 못했습니다. 확인 필요.')
    if len(values) > 100:
        warnings.append('성분 목록이 100개를 초과하여 일부만 읽었습니다. 확인 필요.')
    result['effective_ingredients'] = converted
    cautions = pick(data, 'do_not_take_with', '중복·병용 관련 주의사항')
    if isinstance(cautions, str):
        cautions = [{'item_name': cautions.strip()}] if cautions.strip() else []
    elif isinstance(cautions, list):
        cautions = [dict(v) if isinstance(v, dict) else {'item_name': v} for v in cautions if isinstance(v, (dict,str))]
    else:
        cautions = []
    result['do_not_take_with'] = cautions
    result['schema_source'] = 'korean' if any(k in data for k in ('제품명','유효성분 목록')) else 'english'
    if not result['product_name_raw']:
        warnings.append('제품명을 한 개로 확인하지 못했습니다. 직접 검색하세요.')
    warnings = list(dict.fromkeys(w for w in warnings if isinstance(w, str)))
    result['adapter_warnings'] = warnings
    return result, warnings
```
